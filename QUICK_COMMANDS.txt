===============================================================================
  BORUMITHRA MOBILE - QUICK COMMAND REFERENCE
===============================================================================

📦 INSTALLATION
-------------------------------------------------------------------------------
npm install --legacy-peer-deps                    # Install all dependencies
cd ios && pod install && cd ..                    # Install iOS pods (Mac only)

🔧 CONFIGURATION
-------------------------------------------------------------------------------
cp .env.example .env                              # Create environment file
nano .env                                         # Edit with your keys

# Add these files:
# android/app/google-services.json                # Firebase Android config
# ios/GoogleService-Info.plist                    # Firebase iOS config

🚀 RUN APPLICATION
-------------------------------------------------------------------------------
npm start                                         # Start Metro bundler
npm run android                                   # Run on Android
npm run ios                                       # Run on iOS (Mac only)

npm start & npm run android                       # Start & run Android
npm start & npm run ios                           # Start & run iOS

🧹 CLEANUP COMMANDS
-------------------------------------------------------------------------------
npm start -- --reset-cache                        # Clear Metro cache
rm -rf node_modules package-lock.json             # Remove dependencies
cd android && ./gradlew clean && cd ..            # Clean Android build
cd ios && rm -rf Pods Podfile.lock && cd ..       # Clean iOS pods

🔍 DEBUGGING
-------------------------------------------------------------------------------
npx react-native log-android                      # View Android logs
npx react-native log-ios                          # View iOS logs
npx react-native doctor                           # Check setup
npm run type-check                                # Check TypeScript
npm run lint                                      # Run linter

📱 BUILD FOR PRODUCTION
-------------------------------------------------------------------------------
# Android APK
cd android && ./gradlew assembleRelease && cd ..

# Android AAB (Play Store)
cd android && ./gradlew bundleRelease && cd ..

# iOS
open ios/BorumithraMobile.xcworkspace              # Open in Xcode
# Then: Product → Archive

🐛 COMMON FIXES
-------------------------------------------------------------------------------
# Fix 1: Dependency errors
rm -rf node_modules package-lock.json
npm install --legacy-peer-deps

# Fix 2: Gradle errors
cd android && ./gradlew clean && cd ..
npm run android

# Fix 3: iOS pod errors
cd ios && rm -rf Pods Podfile.lock && pod install && cd ..
npm run ios

# Fix 4: Metro errors
npm start -- --reset-cache

# Fix 5: Build cache
# Android: android/app/build/
# iOS: ios/build/

📊 CHECK VERSIONS
-------------------------------------------------------------------------------
node --version                                    # Should be v18+
npm --version                                     # Should be v9+
npm list react-native                             # Should be 0.73.2
npm list react-native-reanimated                  # Should be ~3.3.0
